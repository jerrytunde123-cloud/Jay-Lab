    // ==================== SHUFFLE ====================
    function shuffleArray(arr) {
      const a = arr.slice();
      for (let i = a.length - 1; i > 0; i--) { const j = Math.floor(Math.random()*(i+1)); [a[i],a[j]]=[a[j],a[i]]; }
      return a;
    }
    function shuffleQuestion(q) {
      const correctText = q.options[q.correct];
      const shuffled = shuffleArray(q.options);
      return { question: q.question, options: shuffled, correct: shuffled.indexOf(correctText) };
    }

    // ==================== BUILD BALANCED POOL ====================
    function buildBalancedPool(subjects) {
      // Get question arrays for each subject
      const pools = {};
      subjects.forEach(subj => {
        const data = quizData[getQuizKey(subj)];
        if (data && data.questions && data.questions.length) {
          pools[subj] = shuffleArray(data.questions).map(q => Object.assign({}, q, { _subject: subj }));
        } else {
          pools[subj] = []; // no questions
        }
      });

      // Target: QUIZ_LENGTH questions total, split equally
      const n = subjects.length;
      const base = Math.floor(QUIZ_LENGTH / n);      // e.g. 20/3 = 6
      const remainder = QUIZ_LENGTH % n;             // e.g. 20%3 = 2

      // Randomly pick `remainder` subjects to get 1 extra question
      const indexes = shuffleArray(subjects.map((_, i) => i));
      const targetCount = {};
      subjects.forEach((s, i) => {
        targetCount[s] = base + (indexes.slice(0, remainder).includes(i) ? 1 : 0);
      });

      let pool = [];
      let leftovers = [];

      subjects.forEach(s => {
        const need = targetCount[s];
        const available = pools[s];
        const take = Math.min(need, available.length);
        pool = pool.concat(available.slice(0, take));
        // Put unused ones in leftovers in case we need more
        leftovers = leftovers.concat(available.slice(take).map(q => Object.assign({}, q, { _fromOverflow: true })));
      });

      // If we still have less than QUIZ_LENGTH (because some subjects had too few),
      // fill from leftovers randomly
      if (pool.length < QUIZ_LENGTH) {
        const shuffledLeftovers = shuffleArray(leftovers);
        const needed = QUIZ_LENGTH - pool.length;
        pool = pool.concat(shuffledLeftovers.slice(0, needed));
      }

      // Final shuffle + limit
      pool = shuffleArray(pool).slice(0, QUIZ_LENGTH);

      // Shuffle each question's options
      return pool.map(shuffleQuestion);
    }

      document.getElementById('scoreLbl').textContent = 'out of ' + questions.length;

      mainScreen.style.display = 'none';
      resultScreen.style.display = 'none';
      quizScreen.style.display = 'block';
      window.scrollTo(0, 0);
      startTimer();
      renderQuestion();
    }

    function startTimer() {
      clearInterval(timerInterval);
      updateTimerDisplay();
      timerInterval = setInterval(() => {
        timeLeft--; updateTimerDisplay();
        if (timeLeft <= 0) { clearInterval(timerInterval); showToast('⏰ Time up!', 'red'); finishQuiz(); }
      }, 1000);
    }
    function updateTimerDisplay() {
      const m = Math.floor(timeLeft/60), s = timeLeft%60;
      const d = document.getElementById('timerDisplay');
      d.innerHTML = '<i class="fas fa-clock"></i> ' + String(m).padStart(2,'0') + ':' + String(s).padStart(2,'0');
      d.classList.toggle('warning', timeLeft <= 60);
    }

    function renderQuestion() {
      const q = currentQuestions[currentIndex];
      document.getElementById('questionCounter').textContent =
        'Question ' + (currentIndex+1) + ' of ' + currentQuestions.length + '  ·  ' + (q._subject || '');
      document.getElementById('questionText').textContent = q.question;
      document.getElementById('progressFill').style.width = ((currentIndex+1)/currentQuestions.length)*100 + '%';

      const letters = ['A','B','C','D','E'];
      const container = document.getElementById('optionsContainer');
      container.innerHTML = '';
      q.options.forEach((opt, i) => {
        const div = document.createElement('div');
        div.className = 'option' + (userAnswers[currentIndex] === i ? ' selected' : '');
        div.innerHTML = '<span class="letter">' + letters[i] + '</span><span>' + opt + '</span>';
        div.addEventListener('click', () => selectOption(i));
        container.appendChild(div);
      });

      document.getElementById('prevBtn').style.display = currentIndex > 0 ? 'block' : 'none';
      const nb = document.getElementById('nextBtn'), sb = document.getElementById('submitBtn');
      if (currentIndex === currentQuestions.length - 1) { nb.style.display = 'none'; sb.style.display = 'block'; }
      else { nb.style.display = 'block'; sb.style.display = 'none'; }
      nb.disabled = userAnswers[currentIndex] === null;
    }
    function selectOption(i) {
      userAnswers[currentIndex] = i;
      document.querySelectorAll('.option').forEach((o, idx) => o.classList.toggle('selected', idx === i));
      document.getElementById('nextBtn').disabled = false;
    }
    document.getElementById('nextBtn').addEventListener('click', () => { if (currentIndex < currentQuestions.length-1) { currentIndex++; renderQuestion(); } });
    document.getElementById('prevBtn').addEventListener('click', () => { if (currentIndex > 0) { currentIndex--; renderQuestion(); } });
    document.getElementById('submitBtn').addEventListener('click', () => {
      const u = userAnswers.filter(a => a === null).length;
      if (u > 0 && !confirm('You have ' + u + ' unanswered question(s). Submit anyway?')) return;
      finishQuiz();
    });

    function finishQuiz() {
      clearInterval(timerInterval);
      let score = 0;
      const letters = ['A','B','C','D','E'];
      const reviewHTML = currentQuestions.map((q, i) => {
        const ok = userAnswers[i] === q.correct;
        if (ok) score++;
        const ua = userAnswers[i] !== null ? letters[userAnswers[i]] + '. ' + q.options[userAnswers[i]] : '<em>Not answered</em>';
        const ca = letters[q.correct] + '. ' + q.options[q.correct];
        return '<div class="review-item ' + (ok ? 'correct' : 'wrong') + '">' +
          '<div class="q">' + (i+1) + '. ' + q.question + ' <span style="color:#0284c7;font-size:0.75rem;">[' + (q._subject||'') + ']</span></div>' +
          '<div class="ans">Your answer: ' + (ok ? '<span class="correct-ans">'+ua+'</span>' : '<span class="user-wrong">'+ua+'</span>') + '</div>' +
          (!ok ? '<div class="ans">Correct: <span class="correct-ans">'+ca+'</span></div>' : '') + '</div>';
      }).join('');

      const total = currentQuestions.length;
      const percent = Math.round((score/total)*100);
      document.getElementById('scoreNum').textContent = score;
      document.getElementById('scoreLbl').textContent = 'out of ' + total;
      document.getElementById('reviewContainer').innerHTML = reviewHTML;

      let title = 'Well done! 🎉', msg = 'Keep practicing to improve your score.';
      if (percent >= 80) { title = 'Excellent! 🌟'; msg = 'You are well prepared for JAMB!'; }
      else if (percent >= 60) { title = 'Good job! 👍'; msg = 'A bit more practice will get you there.'; }
      else if (percent >= 40) { title = 'Keep going! 💪'; msg = 'Review your weak areas and try again.'; }
      else { title = "Don't give up! 📚"; msg = 'Study more and retake the quiz.'; }

      document.getElementById('resultTitle').textContent = title;
      document.getElementById('resultMessage').textContent = msg;
      quizScreen.style.display = 'none';
      resultScreen.style.display = 'block';
      window.scrollTo(0, 0);
      updatePointsUI();
    }

    document.getElementById('restartBtn').addEventListener('click', () => {
      resultScreen.style.display = 'none';
      mainScreen.style.display = 'block';
      selectedSubjects.clear();
      subjectItems.forEach(s => s.classList.remove('selected'));
      updateSelectedUI();
      dailyTopUp();
      updateStartButtonState();
      window.scrollTo(0, 0);
    });

    // ==================== INIT ====================
    function init() {
      getUserId();
      handleIncomingReferral();
      awardReferralToCurrent();
      document.getElementById('refCount').textContent = localStorage.getItem('jamb_ref_count') || '0';
      updatePointsUI();
      checkUnlock();
      updateSelectedUI();
      dailyTopUp();
      updateStartButtonState();
      console.log('✅ JAMB Quiz app ready');
    }
    init();
  })();
</script>
